만들어질 SQL 보기 
-- members see tickets assigned to them or marked shared
create policy tickets_select_member on tickets
  for select
  using (kelay.role() = 'member'
         and (assignee_id = kelay.uid()
              or shared));

-- members update (and move) only tickets assigned to them
create policy tickets_update_member on tickets
  for update
  using (kelay.role() = 'member'
         and assignee_id = kelay.uid())
  with check (assignee_id = kelay.uid());
